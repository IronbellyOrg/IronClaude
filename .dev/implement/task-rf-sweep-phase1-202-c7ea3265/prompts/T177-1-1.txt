+def _iso(value: object) -> str | None:
+    """UTC ISO 8601 with a ``Z`` suffix for an aware UTC datetime; None for anything else."""
+    if value is None or not hasattr(value, "isoformat"):
+        return None
+    return value.isoformat(timespec="seconds").replace("+00:00", "Z")  # type: ignore[attr-defined]

+def _window_text(window: object) -> str:
+    seconds = int(window.total_seconds())  # type: ignore[attr-defined]
+    if seconds > 0 and seconds % 3600 == 0:
+        return f"{seconds // 3600}h"
+    if seconds > 0 and seconds % 60 == 0:
+        return f"{seconds // 60}m"
+    return f"{seconds}s"

+def _yn(value: bool | None) -> str:
+    return "unknown" if value is None else ("yes" if value else "no")

+def _age_text(seconds: int | None) -> str:
+    if seconds is None:
+        return "unknown"
+    if seconds < 60:
+        return f"{seconds}s"
+    if seconds < 3600:
+        return f"{seconds // 60}m"
+    if seconds < 86400:
+        return f"{seconds // 3600}h {seconds % 3600 // 60}m"
+    return f"{seconds // 86400}d {seconds % 86400 // 3600}h"

+def _wait_time(value: object, now: object) -> object | None:
+    """The time in an ``agentWait`` value (a string, or a dict with a known key), else None."""
+    if isinstance(value, dict):
+        times = [
+            t
+            for key in _WAIT_TIME_KEYS
+            if (t := _wait_time(value.get(key), now)) is not None
+        ]
+        return min(times) if times else None
+    if isinstance(value, str):
+        text = value.strip()
+        if text.endswith("Z"):
+            text = text[:-1] + "+00:00"
+        try:
+            parsed = now.fromisoformat(text)  # type: ignore[attr-defined]
+        except ValueError:
+            return None
+        return parsed.replace(tzinfo=now.tzinfo) if parsed.tzinfo is None else parsed  # type: ignore[attr-defined]
+    return None

+def _table(headers: tuple[str, ...], rows: list[tuple[str, ...]]) -> str:
